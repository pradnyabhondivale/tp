import cv2
import numpy as np
import matplotlib.pyplot as plt

img = cv2.imread('hist.jpg', 0)

# RLE Encoding
data = img.flatten()
enc = []
count = 1

for i in range(1, len(data)):
    if data[i] == data[i-1]:
        count += 1
    else:
        enc.append((data[i-1], count))
        count = 1

enc.append((data[-1], count))

# RLE Decoding
dec = []

for value, count in enc:
    dec.extend([value] * count)

out = np.array(dec, dtype=np.uint8).reshape(img.shape)

# Compression size
original = img.size * 8
compressed = len(enc) * 40

print("Compression Ratio:", round(original/compressed, 2))
print("Compressed Size:", compressed, "bits")

# Display
plt.subplot(1,2,1)
plt.imshow(img, cmap='gray')
plt.title("Original")
plt.axis('off')

plt.subplot(1,2,2)
plt.imshow(out, cmap='gray')
plt.title("RLE Output")
plt.axis('off')

plt.show()
