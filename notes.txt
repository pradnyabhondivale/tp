1. Mean filter -----------------------
import cv2
import numpy as np
import matplotlib.pyplot as plt
img = cv2.imread('noisy.jpg', 0)
rows, cols = img.shape
out = img.copy()
for i in range(1, rows-1):
    for j in range(1, cols-1):
        s = 0
        for x in range(-1, 2):
            for y in range(-1, 2):
                s = s + int(img[i+x, j+y])
        out[i, j] = s // 9
plt.subplot(1, 2, 1)
plt.imshow(img, cmap='gray')
plt.title('Input Image')
plt.axis('off')
plt.subplot(1, 2, 2)
plt.imshow(out, cmap='gray')
plt.title('Mean Filter Output')
plt.axis('off')
plt.show()

3. Gaussian Filter  -----------------------
k = [[1,2,1],
     [2,4,2],
     [1,2,1]]

for i in range(1, rows-1):
    for j in range(1, cols-1):
        s = 0
        for x in range(-1, 2):
            for y in range(-1, 2):
                s += int(img[i+x, j+y]) * k[x+1][y+1]
        out[i, j] = s // 16


4.Min Filter
for i in range(1, rows-1):
    for j in range(1, cols-1):
        mn = 255
        for x in range(-1, 2):
            for y in range(-1, 2):
                if img[i+x, j+y] < mn:
                    mn = int(img[i+x, j+y])
        out[i, j] = mn
