2.Histogram Equalization
rows, cols = img.shape
hist = np.zeros(256, dtype=int)

# Histogram count
for i in range(rows):
    for j in range(cols):
        hist[img[i,j]] += 1

# Cumulative histogram
cdf = np.zeros(256, dtype=int)
cdf[0] = hist[0]

for k in range(1,256):
    cdf[k] = cdf[k-1] + hist[k]

# Find minimum CDF
cdf_min = 0
for k in range(256):
    if cdf[k] != 0:
        cdf_min = cdf[k]
        break

# Equalization
out = img.copy()

for i in range(rows):
    for j in range(cols):
        p = int(img[i,j])
        out[i,j] = int((cdf[p]-cdf_min)*255/(rows*cols-cdf_min))

def erosion(a):
    out = np.zeros_like(a)
    for i in range(1, rows-1):
        for j in range(1, cols-1):
            if np.all(a[i-1:i+2, j-1:j+2] == 255):
                out[i,j] = 255
    return out

def dilation(a):
    out = np.zeros_like(a)
    for i in range(1, rows-1):
        for j in range(1, cols-1):
            if np.any(a[i-1:i+2, j-1:j+2] == 255):
                out[i,j] = 255
    return out

2. Region Growing
out = np.zeros_like(img)
visited = np.zeros_like(img)

sx, sy = rows//2, cols//2
seed = img[sx, sy]

stack = [(sx, sy)]
out[sx, sy] = 255
visited[sx, sy] = 1

while stack:
    i, j = stack.pop()
    for x, y in [(i-1,j), (i+1,j), (i,j-1), (i,j+1)]:
        if 0 <= x < rows and 0 <= y < cols:
            if visited[x,y] == 0:
                visited[x,y] = 1
                if abs(int(img[x,y])-int(seed)) < 20:
                    out[x,y] = 255
                    stack.append((x,y))

out = img.copy()
stack = [(0, 0, rows, cols)]
while stack:
    row, col, height, width = stack.pop()
    region = img[row:row+height, col:col+width]
    if height > 8 and width > 8 and (int(region.max())-int(region.min())) > 50:

        half_height = height//2

4. Region Merging
out = img.copy()
block_size = 16
threshold = 25

for row in range(0, rows-block_size, block_size):
    for col in range(0, cols-block_size, block_size):
        region = img[row:row+block_size, col:col+block_size]
        avg = np.mean(region)
        right = img[row:row+block_size, col+block_size:col+2*block_size]
        bottom = img[row+block_size:row+2*block_size, col:col+block_size]
        
        if right.size > 0 and abs(avg-np.mean(right)) < threshold:
            out[row:row+block_size, col:col+block_size] = np.mean([avg, np.mean(right)])
        
        if bottom.size > 0 and abs(avg-np.mean(bottom)) < threshold:
            out[row:row+block_size, col:col+block_size] = np.mean([avg, np.mean(bottom)])

3.Huffman
import cv2
import numpy as np
import matplotlib.pyplot as plt
import heapq

img = cv2.imread('hist.jpg', 0)
data = img.flatten()

# Frequency of each pixel
values, counts = np.unique(data, return_counts=True)

# Create Huffman tree
heap = []
for i in range(len(values)):
    heap.append((counts[i], i, values[i]))

heapq.heapify(heap)
k = len(values)

while len(heap) > 1:
    f1, _, a = heapq.heappop(heap)
    f2, _, b = heapq.heappop(heap)
    heapq.heappush(heap, (f1+f2, k, (a,b)))
    k += 1

tree = heap[0][2]

# Generate codes
codes = {}
stack = [(tree, "")]

while stack:
    node, code = stack.pop()

    if isinstance(node, tuple):
        stack.append((node[1], code+'1'))
        stack.append((node[0], code+'0'))
    else:
        codes[node] = code if code else '0'

# Encoding
enc = ''.join(codes[p] for p in data)

# Decoding
dec = []
code = ""

for bit in enc:
    code += bit
    for value, c in codes.items():
        if code == c:
            dec.append(value)
            code = ""
            break

out = np.array(dec, dtype=np.uint8).reshape(img.shape)

# Compression
original = img.size * 8
compressed = len(enc) + len(values) * 8

print("Compression Ratio:", round(original/compressed, 2))
print("Compressed Size:", compressed, "bits")
